#include <stdio.h>
#include <string.h>

#define max 50
#define size 100
int main()
{
  char backstack[max][size];
  char forwardstack[max][size];
  
  int backtop = -1;
  int forwardtop = -1;
  
  char currentpage[size] = "Home";
  char newpage[size];
  
  int choice;
  do
  {
    printf("\n=====BROWSER NAVIGATION SYSTEM=====\n");
    printf("1.Visit new page\n");
    printf("2.Back navigation\n");
    printf("3.Forward navigation\n");
    printf("4.Display current page\n");
    printf("5.Exit\n");
    
    printf("\n Enter your choice:");
    scanf("%d",&choice); 
    
    switch(choice)
    {
      case 1:
        printf("Enter new page:");
        scanf("%s",newpage);
        
        //Store current page in back stack
        backtop++;
        strcpy(backstack[backtop],currentpage);
        
        //Make new page current
        strcpy(currentpage,newpage);
        
        //Clear forward stack
        forwardtop = -1;
        
        printf("Page visited: %s\n",currentpage);
        break;
        
        case 2:
        //Back navigation
          if(backtop == -1)
          {
            printf("No previous page available\n");
          }
          else
          {
            
            forwardtop++;
            strcpy(forwardstack[forwardtop],currentpage);
            
            
            strcpy(currentpage,backstack[backtop]);
            backtop--;
            printf("Back to: %s\n",currentpage);
          }
          break;
          
          case 3:
          //Forward navigation
          if(forwardtop == -1)
          {
            printf("No forward page available.\n");
          }
          else
          {
            
            backtop++;
            strcpy(backstack[backtop],currentpage);
            
            
            strcpy(currentpage,forwardstack[forwardtop]);
            forwardtop--;
            printf("Forward to: %s\n",currentpage);
          }
          break;
          
          case 4:
          //Display current page
          printf("Current page: %s\n",currentpage);
          break;
          
          case 5:
          //Exiting the system
          printf("Exiting browser navigation system.\n");
          break;
          
          default:
          printf("Invalid choice.\n");
      }
    }while(choice!=5);
  return 0;
}
